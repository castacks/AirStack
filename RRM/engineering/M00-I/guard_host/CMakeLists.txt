cmake_minimum_required(VERSION 3.16)
project(rrm_startup_guard)
set(RRM_UPSTREAM_INCLUDE_DIR "" CACHE PATH "Reviewed AirStack mavros_interface include directory")
if(NOT EXISTS "${RRM_UPSTREAM_INCLUDE_DIR}/mavros_interface/px4_actuation_guard.hpp")
  message(FATAL_ERROR "UNRESOLVED original guard include identity")
endif()
file(SHA256 "${RRM_UPSTREAM_INCLUDE_DIR}/mavros_interface/px4_actuation_guard.hpp" GUARD_SHA)
if(NOT GUARD_SHA STREQUAL "e522a5f6ac10e293e6985b1aaca52d59fad7d80f91a5d5eed0aca0876f785ea2")
  message(FATAL_ERROR "Original guard checksum changed")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(rrm_startup_guard src/guard_host.cpp)
target_compile_features(rrm_startup_guard PRIVATE cxx_std_17)
target_include_directories(rrm_startup_guard PRIVATE "${RRM_UPSTREAM_INCLUDE_DIR}")
ament_target_dependencies(rrm_startup_guard rclcpp ament_index_cpp mavros_msgs rcl_interfaces std_msgs)
target_link_libraries(rrm_startup_guard yaml-cpp)
install(TARGETS rrm_startup_guard DESTINATION lib/${PROJECT_NAME})
ament_package()
